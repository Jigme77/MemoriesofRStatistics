🧪 R code library

📌 Independent t-test
t.test(weight ~ breed, data = cattle, var.equal=TRUE)
📌 ANOVA + Tukey
model <- aov(cortisol ~ housing, data = hens)
TukeyHSD(model)
📌 Kruskal-Wallis
kruskal.test(pain ~ treatment, data = dogs)
📌 Chi-square
chisq.test(table(breed, disease))
📌 Linear regression
lm(milk_yield ~ feed + age, data = cows)
📌 Logistic regression
glm(brd ~ weight + temp, family=binomial, data=calves)
📌 Kaplan-Meier
library(survival)
survfit(Surv(time,status) ~ treatment)
